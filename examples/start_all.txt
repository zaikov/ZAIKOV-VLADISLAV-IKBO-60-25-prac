# Стартовый скрипт для проверки всех команд этапов 1–3.
# ВНИМАНИЕ: vfs-init очищает CSV-файл VFS, поэтому запускайте скрипт
# с копией VFS (так делает scripts/test_vfs.sh).

# --- ls: заглушка, все допустимые варианты ключей
ls
ls /home/user
ls -l
ls -a /etc
ls -la /home /var/log
ls -al -l

# --- cd: заглушка
cd
cd /home/user/docs
cd ..

# --- vfs-init: замена VFS на VFS по умолчанию и очистка CSV-файла
vfs-init

# --- обработка ошибок: скрипт остановится на следующей строке,
# а команды после неё не выполнятся
vfs-init extra
ls
